```python
"""Unit tests for VideoManager — no real browser or ffmpeg required.

Covers VID-01 through VID-07:
  VID-01  VideoManager interface shape
  VID-02  AppConfig.record_video field
  VID-03  headless guard returns None
  VID-04  ffmpeg-absent guard returns None
  VID-05  delete() removes file
  VID-06  VIDEO_DIR / VIDEO_DATE_FORMAT constants
  VID-07  fixture logic: delete on pass, retain on fail (via monkeypatch)
"""
from __future__ import annotations

import os
import subprocess
from pathlib import Path
from typing import Optional
from unittest.mock import MagicMock, patch

import pytest

from src.core.config import AppConfig
from src.core.constants import VIDEO_DATE_FORMAT, VIDEO_DIR
from src.utils.videos import VideoManager


# ---------------------------------------------------------------------------
# VID-06: Constants
# ---------------------------------------------------------------------------

class TestVideoConstants:
    """VIDEO_DIR and VIDEO_DATE_FORMAT have correct values."""

    def test_video_dir_value(self):
        assert VIDEO_DIR == "reports/videos"

    def test_video_date_format_value(self):
        assert VIDEO_DATE_FORMAT == "%Y%m%d_%H%M%S"


# ---------------------------------------------------------------------------
# VID-01: VideoManager interface
# ---------------------------------------------------------------------------

class TestVideoManagerInterface:
    """VideoManager exposes the required interface."""

    def test_has_start_method(self):
        assert callable(VideoManager.start)

    def test_has_stop_method(self):
        assert callable(VideoManager.stop)

    def test_has_delete_method(self):
        assert callable(VideoManager.delete)

    def test_init_sets_base_dir(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        assert m._base_dir == tmp_path

    def test_init_default_base_dir(self):
        m = VideoManager()
        assert str(m._base_dir) == VIDEO_DIR


# ---------------------------------------------------------------------------
# VID-02: AppConfig.record_video field
# ---------------------------------------------------------------------------

class TestAppConfigRecordVideo:
    """AppConfig.record_video defaults to False, reads from YAML and env var."""

    def _yaml(self, tmp_path, data: dict) -> str:
        import yaml
        f = tmp_path / "env.test.yaml"
        f.write_text(yaml.dump(data), encoding="utf-8")
        return str(tmp_path)

    def test_record_video_defaults_to_false(self, tmp_path):
        config = AppConfig(env="test", config_dir=str(tmp_path))
        assert config.record_video is False

    def test_record_video_true_from_yaml(self, tmp_path):
        config_dir = self._yaml(tmp_path, {"record_video": True})
        config = AppConfig(env="test", config_dir=config_dir)
        assert config.record_video is True

    def test_record_video_false_from_yaml(self, tmp_path):
        config_dir = self._yaml(tmp_path, {"record_video": False})
        config = AppConfig(env="test", config_dir=config_dir)
        assert config.record_video is False

    def test_record_video_true_from_env_var(self, tmp_path, monkeypatch):
        monkeypatch.setenv("RECORD_VIDEO", "true")
        config = AppConfig(env="test", config_dir=str(tmp_path))
        assert config.record_video is True

    def test_record_video_env_var_overrides_yaml(self, tmp_path, monkeypatch):
        config_dir = self._yaml(tmp_path, {"record_video": False})
        monkeypatch.setenv("RECORD_VIDEO", "true")
        config = AppConfig(env="test", config_dir=config_dir)
        assert config.record_video is True


# ---------------------------------------------------------------------------
# VID-03: Headless guard
# ---------------------------------------------------------------------------

class TestVideoManagerHeadless:
    """VideoManager.start() returns None when headless=True."""

    def test_start_returns_none_when_headless(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        result = m.start("test_headless", headless=True)
        assert result is None

    def test_no_subprocess_spawned_when_headless(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        with patch("subprocess.Popen") as mock_popen:
            m.start("test_headless", headless=True)
            mock_popen.assert_not_called()

# VID-04: ffmpeg-absent guard
# ---------------------------------------------------------------------------

class TestVideoManagerNoFfmpeg:
    """VideoManager.start() returns None when ffmpeg is not on PATH."""

    def test_start_returns_none_when_ffmpeg_absent(self, tmp_path, monkeypatch):
        monkeypatch.setattr("shutil.which", lambda x: None)
        m = VideoManager(base_dir=str(tmp_path))
        result = m.start("test_no_ffmpeg")
        assert result is None

    def test_no_subprocess_spawned_when_ffmpeg_absent(self, tmp_path, monkeypatch):
        monkeypatch.setattr("shutil.which", lambda x: None)
        m = VideoManager(base_dir=str(tmp_path))
        with patch("subprocess.Popen") as mock_popen:
            m.start("test_no_ffmpeg")
            mock_popen.assert_not_called()


# ---------------------------------------------------------------------------
# VID-05: delete()
# ---------------------------------------------------------------------------

class TestVideoManagerDelete:
    """VideoManager.delete() removes the file; no-ops on missing file."""

    def test_delete_removes_existing_file(self, tmp_path):
        f = tmp_path / "test_video.mp4"
        f.write_bytes(b"fake mp4 content")
        assert f.exists()
        m = VideoManager(base_dir=str(tmp_path))
        m.delete(str(f))
        assert not f.exists()

    def test_delete_no_op_on_missing_file(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        # Should not raise
        m.delete(str(tmp_path / "nonexistent.mp4"))

    def test_delete_only_removes_target_file(self, tmp_path):
        f1 = tmp_path / "keep.mp4"
        f2 = tmp_path / "delete.mp4"
        f1.write_bytes(b"keep")
        f2.write_bytes(b"delete")
        m = VideoManager(base_dir=str(tmp_path))
        m.delete(str(f2))
        assert f1.exists()
        assert not f2.exists()


# ---------------------------------------------------------------------------
# stop() behavior
# ---------------------------------------------------------------------------

class TestVideoManagerStop:
    """VideoManager.stop() is a no-op when no process; sends 'q' when running."""

    def test_stop_no_op_when_no_process(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        # Should not raise
        m.stop()
        assert m._proc is None

    def test_stop_sends_q_to_stdin(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        mock_proc = MagicMock()
        mock_proc.poll.return_value = None
        mock_proc.stdin = MagicMock()
        m._proc = mock_proc
        m.stop()
        mock_proc.stdin.write.assert_called_once_with(b"q")
        mock_proc.stdin.flush.assert_called_once()

    def test_stop_kills_on_timeout(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        mock_proc = MagicMock()
        mock_proc.poll.return_value = None
        mock_proc.stdin = MagicMock()
        mock_proc.wait.side_effect = subprocess.TimeoutExpired(cmd="ffmpeg", timeout=10)
        m._proc = mock_proc
        m.stop()
        mock_proc.kill.assert_called_once()

    def test_stop_no_op_when_process_already_terminated(self, tmp_path):
        m = VideoManager(base_dir=str(tmp_path))
        mock_proc = MagicMock()
        mock_proc.poll.return_value = 0  # already exited
        m._proc = mock_proc
        m.stop()
        mock_proc.stdin.write.assert_not_called()


# ---------------------------------------------------------------------------
# VID-07: Fixture delete-on-pass / retain-on-fail logic
# (Tests the decision logic in isolation using monkeypatched VideoManager)
# ---------------------------------------------------------------------------

class TestVideoFixtureDeleteLogic:
    """The keep/delete decision made by video_recorder teardown logic."""

    def _make_manager_with_delete_tracker(self, tmp_path) -> tuple:
        """Return (manager, video_path, deleted_paths_list)."""
        f = tmp_path / "recording.mp4"
        f.write_bytes(b"fake")
        m = VideoManager(base_dir=str(tmp_path))
        deleted = []
        original_delete = m.delete

        def tracking_delete(path):
            deleted.append(path)
            original_delete(path)

        m.delete = tracking_delete
        return m, str(f), deleted

    def test_delete_called_when_test_passes(self, tmp_path):
        manager, video_path, deleted = self._make_manager_with_delete_tracker(tmp_path)
        # Simulate: test passed — call_report.failed = False
        test_failed = False
        if video_path:
            if test_failed:
                pass  # retain
            else:
                manager.delete(video_path)
        assert deleted == [video_path]
        assert not Path(video_path).exists()

    def test_delete_not_called_when_test_fails(self, tmp_path):
        manager, video_path, deleted = self._make_manager_with_delete_tracker(tmp_path)
        # Simulate: test failed — call_report.failed = True
        test_failed = True
        if video_path:
            if test_failed:
                pass  # retain
            else:
                manager.delete(video_path)
        assert deleted == []
        assert Path(video_path).exists()

    def test_no_delete_when_video_path_is_none(self, tmp_path):
        manager, _, _ = self._make_manager_with_delete_tracker(tmp_path)
        deleted = []
        video_path = None  # recording was skipped
        if video_path:
            manager.delete(video_path)
        assert deleted == []
```
